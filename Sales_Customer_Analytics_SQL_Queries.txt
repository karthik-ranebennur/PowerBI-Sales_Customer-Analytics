SALES_ORDER_DETAILS

This is the transaction/line-item table.

CREATE TABLE SALES_ORDER_DETAILS (
    ORDER_DETAIL_ID NUMBER PRIMARY KEY,
    ORDER_ID NUMBER NOT NULL,
    PRODUCT_ID NUMBER NOT NULL,
    QUANTITY NUMBER NOT NULL,
    UNIT_PRICE NUMBER(10,2) NOT NULL,
    DISCOUNT_PERCENT NUMBER(5,2) DEFAULT 0
);

Add order foreign key

ALTER TABLE SALES_ORDER_DETAILS
ADD CONSTRAINT FK_DETAIL_ORDER
FOREIGN KEY (ORDER_ID)
REFERENCES SALES_ORDERS(ORDER_ID);

Add product foreign key

ALTER TABLE SALES_ORDER_DETAILS
ADD CONSTRAINT FK_DETAIL_PRODUCT
FOREIGN KEY (PRODUCT_ID)
REFERENCES SALES_PRODUCTS(PRODUCT_ID);

Generate 50,000 details

INSERT INTO SALES_ORDER_DETAILS
(
    ORDER_DETAIL_ID,
    ORDER_ID,
    PRODUCT_ID,
    QUANTITY,
    UNIT_PRICE,
    DISCOUNT_PERCENT
)
SELECT
    100000 + ROW_NUMBER() OVER (ORDER BY ORDER_ID, PRODUCT_ID),
    ORDER_ID,
    PRODUCT_ID,
    QUANTITY,
    UNIT_PRICE,
    DISCOUNT_PERCENT
FROM
(
    SELECT
        o.ORDER_ID,
        p.PRODUCT_ID,
        TRUNC(DBMS_RANDOM.VALUE(1, 6)) AS QUANTITY,
        p.UNIT_PRICE,
        ROUND(DBMS_RANDOM.VALUE(0, 16), 2)
            AS DISCOUNT_PERCENT
    FROM SALES_ORDERS o
    CROSS JOIN SALES_PRODUCTS p
    WHERE p.PRODUCT_ID IN
    (
        101,102,103,104,105,106,107,108,109,110,
        111,112,113,114,115,116,117,118,119,120
    )
)
WHERE MOD(ORDER_ID + PRODUCT_ID, 8) IN (0,1);

COMMIT;

Important point

This generated:

10,000 orders × 5 product lines = 50,000 details

So every order currently has 5 product lines.


SALES_TARGETS

CREATE TABLE SALES_TARGETS (
    TARGET_ID NUMBER PRIMARY KEY,
    TARGET_MONTH DATE NOT NULL,
    REGION_ID NUMBER NOT NULL,
    TARGET_AMOUNT NUMBER(15,2) NOT NULL
);

Generate targets

INSERT INTO SALES_TARGETS
(
    TARGET_ID,
    TARGET_MONTH,
    REGION_ID,
    TARGET_AMOUNT
)
SELECT
    300000 + ROW_NUMBER() OVER
        (ORDER BY REGION_ID, TARGET_MONTH),
    TARGET_MONTH,
    REGION_ID,
    ROUND(
        500000 + DBMS_RANDOM.VALUE(0, 500000),
        2
    )
FROM
(
    SELECT
        ADD_MONTHS(
            DATE '2023-01-01',
            LEVEL - 1
        ) AS TARGET_MONTH
    FROM DUAL
    CONNECT BY LEVEL <= 48
) M
CROSS JOIN SALES_REGIONS R;

COMMIT;

48 months × 16 regions:

DATA VALIDATION QUERIES

Check invalid customers

SELECT COUNT(*) AS INVALID_CUSTOMER_ORDERS
FROM SALES_ORDERS O
LEFT JOIN SALES_CUSTOMERS C
    ON O.CUSTOMER_ID = C.CUSTOMER_ID
WHERE C.CUSTOMER_ID IS NULL;

------------------------------------------------------------------

Check invalid order details

SELECT COUNT(*) AS INVALID_ORDER_DETAILS
FROM SALES_ORDER_DETAILS D
LEFT JOIN SALES_ORDERS O
    ON D.ORDER_ID = O.ORDER_ID
WHERE O.ORDER_ID IS NULL;

----------------------------------------------------------------

Check invalid products

SELECT COUNT(*) AS INVALID_PRODUCT_DETAILS
FROM SALES_ORDER_DETAILS D
LEFT JOIN SALES_PRODUCTS P
    ON D.PRODUCT_ID = P.PRODUCT_ID
WHERE P.PRODUCT_ID IS NULL;

----------------------------------------------------------------
Check invalid quantity, price and discount

SELECT
    COUNT(*) AS INVALID_ROWS
FROM SALES_ORDER_DETAILS
WHERE QUANTITY <= 0
   OR UNIT_PRICE <= 0
   OR DISCOUNT_PERCENT < 0
   OR DISCOUNT_PERCENT > 100;

ORDER STATUS VALIDATION

SOURCE-OF-TRUTH — TOTAL SALES

This is extremely important for our Power BI validation.

SELECT
    ROUND(
        SUM(
            QUANTITY * UNIT_PRICE *
            (1 - DISCOUNT_PERCENT / 100)
        ),
        2
    ) AS TOTAL_SALES
FROM SALES_ORDER_DETAILS;


Actual result

₹2,400,949,378

≈ ₹240.09 Crore

====================================================================
SOURCE-OF-TRUTH — TOTAL COST

SELECT
    ROUND(
        SUM(D.QUANTITY * P.COST_PRICE),
        2
    ) AS TOTAL_COST
FROM SALES_ORDER_DETAILS D
JOIN SALES_PRODUCTS P
    ON D.PRODUCT_ID = P.PRODUCT_ID;

Actual result

₹2,030,296,180

====================================================================
SOURCE-OF-TRUTH — TOTAL PROFIT


SELECT
    ROUND(
        SUM(
            D.QUANTITY * D.UNIT_PRICE
                * (1 - D.DISCOUNT_PERCENT / 100)
            - D.QUANTITY * P.COST_PRICE
        ),
        2
    ) AS TOTAL_PROFIT
FROM SALES_ORDER_DETAILS D
JOIN SALES_PRODUCTS P
    ON D.PRODUCT_ID = P.PRODUCT_ID;

Actual result

₹370,653,198
